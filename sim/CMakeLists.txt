cmake_minimum_required(VERSION 3.26)
project(MNISTDramSim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
file(GLOB DRAM_SOURCES "${CMAKE_CURRENT_LIST_DIR}/../third_party/DRAMSim2/*.cpp")
list(FILTER DRAM_SOURCES EXCLUDE REGEX "/TraceBasedSim\\.cpp$")
add_library(dramsim STATIC ${DRAM_SOURCES})
set_target_properties(dramsim PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(dramsim PRIVATE NO_STORAGE)
target_compile_options(dramsim PRIVATE -O3)
target_include_directories(dramsim PUBLIC "${CMAKE_CURRENT_LIST_DIR}/../third_party/DRAMSim2")
