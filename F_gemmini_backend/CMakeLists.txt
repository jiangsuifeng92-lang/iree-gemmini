# Build the tested H steps 6/7/8 backend using one vendored IREE/LLVM copy.
cmake_minimum_required(VERSION 3.26)
project(MNISTGemminiBackend LANGUAGES C CXX)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build mode")
set(IREE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../third_party/iree" CACHE PATH "Vendored IREE source")
set(IREE_BUNDLED_LLVM_DIR "${CMAKE_CURRENT_LIST_DIR}/../third_party/llvm-project" CACHE PATH "Vendored patched LLVM")
if(NOT EXISTS "${IREE_SOURCE_DIR}/compiler/CMakeLists.txt" OR NOT EXISTS "${IREE_BUNDLED_LLVM_DIR}/llvm/CMakeLists.txt")
  message(FATAL_ERROR "Vendored IREE/LLVM source is missing")
endif()
set(LLVM_EXTERNAL_LIT "${IREE_BUNDLED_LLVM_DIR}/llvm/utils/lit/lit.py" CACHE FILEPATH "" FORCE)
set(IREE_DEFAULT_CPU_LLVM_TARGETS "RISCV;X86" CACHE STRING "Needed device and host targets" FORCE)
set(IREE_INPUT_STABLEHLO ON CACHE BOOL "" FORCE)
set(IREE_INPUT_TORCH OFF CACHE BOOL "" FORCE)
set(IREE_INPUT_TOSA OFF CACHE BOOL "" FORCE)
set(IREE_TARGET_BACKEND_VMVX OFF CACHE BOOL "" FORCE)
set(LLVM_PARALLEL_LINK_JOBS 1 CACHE STRING "Limit compiler link memory")
set(IREE_CMAKE_PLUGIN_PATHS "${CMAKE_CURRENT_LIST_DIR}/build_tools/cmake")
set(IREE_BUILD_COMPILER ON CACHE BOOL "" FORCE)
set(IREE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_PYTHON_BINDINGS OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_BINDINGS_PYTHON OFF CACHE BOOL "" FORCE)
set(IREE_TARGET_BACKEND_DEFAULTS OFF CACHE BOOL "" FORCE)
set(IREE_TARGET_BACKEND_LLVM_CPU ON CACHE BOOL "" FORCE)
set(IREE_HAL_DRIVER_DEFAULTS OFF CACHE BOOL "" FORCE)
set(IREE_HAL_DRIVER_LOCAL_SYNC ON CACHE BOOL "" FORCE)
set(IREE_HAL_DRIVER_LOCAL_TASK OFF CACHE BOOL "" FORCE)
set(IREE_ENABLE_ASSERTIONS ON CACHE BOOL "" FORCE)
set(IREE_ENABLE_LLD ON CACHE BOOL "" FORCE)
set(IREE_ENABLE_LIBBACKTRACE OFF CACHE BOOL "" FORCE)
set(IREE_ENABLE_SPLIT_DWARF OFF CACHE BOOL "" FORCE)
set(IREE_ENABLE_THIN_ARCHIVES OFF CACHE BOOL "" FORCE)
set(IREE_ERROR_ON_MISSING_SUBMODULES OFF CACHE BOOL "" FORCE)
add_subdirectory("${IREE_SOURCE_DIR}" third_party/iree)
